cmake_minimum_required(VERSION 3.22.1)
project(soundbooki_g2p)
add_library(sea_g2p_rs SHARED IMPORTED)
set_target_properties(sea_g2p_rs PROPERTIES IMPORTED_LOCATION ${CMAKE_CURRENT_SOURCE_DIR}/../jniLibs/${ANDROID_ABI}/libsea_g2p_rs.so)
set_target_properties(sea_g2p_rs PROPERTIES IMPORTED_NO_SONAME TRUE)
add_library(soundbooki_g2p SHARED bridge.cpp)
target_include_directories(soundbooki_g2p PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/sea-g2p/include)
target_link_libraries(soundbooki_g2p sea_g2p_rs)
target_link_options(soundbooki_g2p PRIVATE "-Wl,-z,max-page-size=16384")
